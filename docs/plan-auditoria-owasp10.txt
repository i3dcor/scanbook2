│ HAL-02 — Validación de ISBN insuficiente (MEDIO)                             │
│                                                                              │
│ Archivo: domain/model/ScannedIsbn.kt,                                        │
│ presentation/viewmodel/EditBookViewModel.kt                                  │
│                                                                              │
│ El ISBN se acepta sin verificar el dígito de control (checksum Luhn para     │
│ ISBN-13).                                                                    │
│ Un QR malicioso puede inyectar strings arbitrarios que llegan a la BD y al   │
│ nombre de archivo.                                                           │
│                                                                              │
│ Mitigación:                                                                  │
│ fun isValidIsbn13(isbn: String): Boolean {                                   │
│     if (isbn.length != 13 || !isbn.all { it.isDigit() }) return false        │
│     val sum = isbn.dropLast(1).mapIndexed { i, c ->                          │
│         c.digitToInt() * if (i % 2 == 0) 1 else 3                            │
│     }.sum()                                                                  │
│     return (10 - sum % 10) % 10 == isbn.last().digitToInt()                  │
│ }                                                                            │
